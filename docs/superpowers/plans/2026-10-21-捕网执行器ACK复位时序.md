# 捕网执行器 ACK 复位时序 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 触发命令收到 PX4 ACK ACCEPTED 后保持 1.5s 再复位，复位 ACK + 稳定时间后 Ready；ACK 契约破坏时停止任务控制并交 PX4/遥控；为多发预留接口。

**Architecture:** 时序状态机放在 `SinglePwmNetActuator` 内部；状态机只做“每拍 Update + S10 触发 + 故障处置”。ACK 状态经 `common::CommandAckStatus` 从 `IPx4CommandSender` 读取；新增逻辑集中在执行器小文件，`mission_state_machine.cpp` 只增加薄集成层。

**Tech Stack:** C++17、CMake、Google Test、MAVLink c_library_v2、PX4 1.17.0、WSL2 Ubuntu 24.04。

**Spec:** `docs/superpowers/specs/2026-10-21-捕网执行器ACK复位时序-design.md`

## Global Constraints

- 所有编译/测试在 WSL2 Ubuntu 24.04 中进行；提交前必须编译通过。
- 沟通、注释、提交说明、文档统一中文；标识符英文。

## Review Focus

| 风险输入/条件 | 期望行为 | 归属任务 |
|---|---|---|
| ACK 在超时边界之后到达 | 已 Faulted，不得再推进/发复位 | Task 2/4 |
| IN_PROGRESS 后最终 ACCEPTED | 保持等待，最终 ACCEPTED 才启动保持计时 | Task 4 |
| 触发 ACK 后、复位前断连 | Faulted，不发复位；状态机 S16 | Task 4/5 |
| 复位入队失败但链路仍连接 | 下拍重试复位，不混入触发命令 | Task 4/5 |
| 窗口中收到无关命令 ACK（如 176） | 忽略，不推进、不故障 | Task 4 |

#pragma once

#include <array>
#include <cstdint>
#include <limits>
#include <mutex>
#include <vector>

#include "common/types.h"
#include "communication/px4_link.h"

namespace drone::test {

class FakeCommandSender final : public communication::IPx4CommandSender {
public:
    struct CallRecord {
        uint16_t command = 0;
        std::array<float, 7> params{};
        bool queued = false;
    };

    bool SendCommand(uint16_t mavlink_command, float param1, float param2,
                     float param3, float param4, float param5, float param6,
                     float param7) override {
        std::lock_guard<std::mutex> lock(mutex_);
        ++call_count_;
        last_command_ = mavlink_command;
        last_params_ = {param1, param2, param3, param4, param5, param6, param7};
        const bool queued = fail_calls_remaining_ == 0;
        if (!queued) {
            --fail_calls_remaining_;
        }
        history_.push_back({mavlink_command, last_params_, queued});
        return queued;
    }

    uint64_t AckMatchCount() const override {
        std::lock_guard<std::mutex> lock(mutex_);
        return ack_.matched_count;
    }

    uint64_t AckTimeoutCount() const override {
        std::lock_guard<std::mutex> lock(mutex_);
        return ack_.timeout_count;
    }

    common::CommandAckStatus AckStatus() const override {
        std::lock_guard<std::mutex> lock(mutex_);
        return ack_;
    }

    void CancelPendingCommands() override {
        std::lock_guard<std::mutex> lock(mutex_);
        ++cancel_count_;
    }

    void SimulateAck(uint16_t command, uint8_t result, uint64_t time_ms) {
        std::lock_guard<std::mutex> lock(mutex_);
        ++ack_.matched_count;
        ack_.last_command = command;
        ack_.last_result = result;
        ack_.last_time_ms = time_ms;
    }

    void SimulateTimeout(uint16_t command) {
        std::lock_guard<std::mutex> lock(mutex_);
        ++ack_.timeout_count;
        ack_.last_timeout_command = command;
    }

    void FailNextCalls(uint64_t count) {
        std::lock_guard<std::mutex> lock(mutex_);
        fail_calls_remaining_ = count;
    }

    uint64_t CallCount() const {
        std::lock_guard<std::mutex> lock(mutex_);
        return call_count_;
    }

    uint64_t CancelCount() const {
        std::lock_guard<std::mutex> lock(mutex_);
        return cancel_count_;
    }

    uint16_t LastCommand() const {
        std::lock_guard<std::mutex> lock(mutex_);
        return last_command_;
    }

    std::array<float, 7> LastParams() const {
        std::lock_guard<std::mutex> lock(mutex_);
        return last_params_;
    }

    std::vector<CallRecord> History() const {
        std::lock_guard<std::mutex> lock(mutex_);
        return history_;
    }

private:
    mutable std::mutex mutex_;
    common::CommandAckStatus ack_;
    uint64_t call_count_ = 0;
    uint64_t cancel_count_ = 0;
    uint64_t fail_calls_remaining_ = 0;
    uint16_t last_command_ = 0;
    std::array<float, 7> last_params_{};
    std::vector<CallRecord> history_;
};

}  // namespace drone::test

```cpp
class SinglePwmNetActuator final : public INetActuator {
public:
    SinglePwmNetActuator(config::ActuatorConfig config,
                         communication::IPx4CommandSender* sender)
        : config_(std::move(config)), sender_(sender) {}

    bool Trigger() override {
        if (state_ != State::kReady || remaining_shots_ == 0) {
            return false;
        }
        if (!SendNorm(config_.net_fire_norm, "触发")) {
            return false;
        }
        remaining_shots_ = 0;
        fired_ = true;
        BeginWait(State::kFireQueued);
        return true;
    }

    bool Reset() override {
        if (Faulted()) {
            return false;
        }
        if (!SendNorm(config_.net_idle_norm, "复位行程")) {
            return false;
        }
        state_ = State::kReady;
        return true;
    }

    void Update(const ActuatorFeedback& feedback) override {
        if (state_ == State::kFaulted || state_ == State::kReady) {
            return;
        }
        if (ProcessAck(feedback)) {
            return;
        }
        if (!feedback.px4_connected &&
            (state_ == State::kFireQueued || state_ == State::kHolding ||
             state_ == State::kResetQueued)) {
            EnterFault("等待 PX4 ACK 期间连接断开");
            return;
        }
        if (state_ == State::kFireAcked) {
            state_ = State::kHolding;
        }
        if (state_ == State::kHolding && feedback.now_ms >= hold_deadline_ms_) {
            if (!SendNorm(config_.net_idle_norm, "复位行程")) {
                return;  // 连接正常但队列满：下拍重试
            }
            BeginWait(State::kResetQueued);
            return;
        }
        if (state_ == State::kResetAcked) {
            state_ = State::kSettling;
        }
        if (state_ == State::kSettling &&
            feedback.now_ms >= settle_deadline_ms_) {
            state_ = State::kReady;
        }
    }

    bool Ready() const override { return state_ == State::kReady; }
    uint32_t RemainingShots() const override { return remaining_shots_; }
    bool Fired() const override { return fired_; }
    bool Faulted() const override { return state_ == State::kFaulted; }
    const char* FaultReason() const override { return fault_reason_; }
    const char* Type() const override { return "single_pwm"; }

private:
    enum class State {
        kReady,
        kFireQueued,
        kFireAcked,
        kHolding,
        kResetQueued,
        kResetAcked,
        kSettling,
        kFaulted,
    };

    void BeginWait(State next) {
        const auto ack = sender_->AckStatus();
        ack_baseline_ = ack.matched_count;
        timeout_baseline_ = ack.timeout_count;
        state_ = next;
    }

    bool ProcessAck(const ActuatorFeedback& feedback) {
        const auto& ack = feedback.ack;
        const bool matched_new = ack.matched_count != ack_baseline_;
        const bool timeout_new = ack.timeout_count != timeout_baseline_;
        ack_baseline_ = ack.matched_count;
        timeout_baseline_ = ack.timeout_count;

        if (timeout_new && ack.last_timeout_command == kMavCmdDoSetActuator) {
            EnterFault("等待 187 ACK 超时");
            return true;
        }
        if (!matched_new || ack.last_command != kMavCmdDoSetActuator) {
            return false;
        }
        if (ack.last_result == kMavResultInProgress) {
            return false;
        }
        if (ack.last_result != kMavResultAccepted) {
            EnterFault("187 ACK 结果非 ACCEPTED");
            return true;
        }

        const uint64_t ack_time =
            ack.last_time_ms == 0 ? feedback.now_ms : ack.last_time_ms;
        if (state_ == State::kFireQueued) {
            state_ = State::kFireAcked;
            hold_deadline_ms_ = ack_time +
                static_cast<uint64_t>(config_.reset_delay.count());
        } else if (state_ == State::kResetQueued) {
            state_ = State::kResetAcked;
            settle_deadline_ms_ = ack_time +
                static_cast<uint64_t>(config_.reset_settle.count());
        }
        return false;
    }

    bool SendNorm(double normalized, const char* action) {
        if (sender_ == nullptr) {
            return false;
        }
        const int32_t channel = config_.net_channel;
        if (channel < 1 || channel > 6) {
            EnterFault("执行器通道越界");
            return false;
        }
        std::array<float, 7> params;
        params.fill(std::numeric_limits<float>::quiet_NaN());
        params[static_cast<std::size_t>(channel - 1)] =
            static_cast<float>(normalized);
        params[6] = 0.0F;
        const bool queued = sender_->SendCommand(
            kMavCmdDoSetActuator, params[0], params[1], params[2], params[3],
            params[4], params[5], params[6]);
        if (queued) {
            SPDLOG_WARN("捕网执行器{}命令已入队: type={} set={} norm={:.2f}",
                        action, Type(), channel, normalized);
            return true;
        }
        const uint64_t count = ++failure_count_;
        if (count == 1 || count % 100 == 0) {
            SPDLOG_WARN("捕网执行器{}命令入队失败（下拍重试）: type={} set={} "
                        "count={}", action, Type(), channel, count);
        }
        return false;
    }

    void EnterFault(const char* reason) {
        if (state_ == State::kFaulted) {
            return;
        }
        state_ = State::kFaulted;
        fault_reason_ = reason;
        SPDLOG_ERROR("捕网执行器故障: type={} set={} reason={}",
                     Type(), config_.net_channel, reason);
    }

    const config::ActuatorConfig config_;
    communication::IPx4CommandSender* const sender_;
    State state_ = State::kReady;
    uint32_t remaining_shots_ = 1;
    bool fired_ = false;
    uint64_t ack_baseline_ = 0;
    uint64_t timeout_baseline_ = 0;
    uint64_t hold_deadline_ms_ = 0;
    uint64_t settle_deadline_ms_ = 0;
    uint64_t failure_count_ = 0;
    const char* fault_reason_ = "";
};
```
