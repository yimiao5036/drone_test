## Review Focus

以下输入/失效模式 spec 隐含但容易被漏测，已在对应任务中钉测试：

1. **意图字段含 NaN/Inf**（地面站异常或上游缺陷）→ FlightController 拒绝并走降级出口（Task 3 步骤 1）。
2. **意图到达即超龄**（`valid_until_ms < now`，上游时钟错配）→ 视同超龄，走 COAST 刹车而非执行（Task 3 步骤 1）。
3. **control_source 出现未定义值**（协议扩展后旧代码）→ 不崩溃、按安全来源处理并节流 WARN（Task 4 步骤 1）。
4. **PX4 模式在 Offboard/非 Offboard 间快速抖动**（飞手犹豫拨杆）→ 停发/恢复判定不抖动：离开即 IDLE，回 ACTIVE 需授权链重新全通（Task 5 步骤 1）。
5. **单调时钟回退**（now < 上一周期）→ dt<=0 时限幅器冻结输出（SlewRateLimiter 已有该语义），不得产生反向跳变（Task 4 步骤 1 回归）。

    if (period.count() <= 0) throw std::invalid_argument("controller.period_ms 必须为正");
    if (max_speed_mps <= 0.0) throw std::invalid_argument("controller.max_speed_mps 必须为正");
    if (max_accel_mps2 <= 0.0) throw std::invalid_argument("controller.max_accel_mps2 必须为正");
    if (max_jerk_mps3 < 0.0) throw std::invalid_argument("controller.max_jerk_mps3 不得为负");
    if (max_altitude_m <= 0.0 || max_radius_m <= 0.0)
        throw std::invalid_argument("controller 围栏参数必须为正");
    if (intent_timeout.count() <= 0)

    if (coast_timeout.count() <= intent_timeout.count())

    if (slew_transition.count() < 0)

```cpp
// 固定周期线程：每 period 执行一次 Tick()
void Run() {
    auto next = Clock::now();
    while (running_) {
        next += config_.period;
        Tick(MonotonicMs());
        std::this_thread::sleep_until(next);
    }
}

void Tick(uint64_t now_ms) {
    // 1) 排干意图队列，只留最新（Topic 订阅队列有界，TryTake 循环）
    DrainIntents();
    // 2) 门禁链：IDLE 直接返回（本任务默认 IDLE，Task 5 才接入授权）
    if (authority_ == ControlAuthority::kIdle) return;
    if (!latest_intent_.has_value()) return;
    // ③ 有限性
    if (!IsFiniteIntent(*latest_intent_)) { ++rejected_intent_count_; latest_intent_.reset(); return; }
    // ① 超龄（含到达即超龄）
    if (now_ms >= latest_intent_->valid_until_ms ||
        StalenessChecker::IsStale(static_cast<int64_t>(latest_intent_->header.receive_time_ms),
                                  static_cast<int64_t>(now_ms), config_.intent_timeout.count())) {
        ++rejected_intent_count_;
        EnterCoastOrIdle(now_ms);   // Task 5 完整实现；本任务先发布零速刹车
        PublishBrake();
        return;
    }
    PublishFromIntent(*latest_intent_);  // Task 4 填限幅；本任务先直通
}
```

`IsFiniteIntent` 对 `target_x/y/z/yaw_deg/yaw_rate_dps/max_speed_mps` 全部 `std::isfinite`。
`PublishBrake()` 发布 `type=kVelocityHeading`、三分量 0、`valid=true` 的 setpoint。
Start/Stop 幂等照 Stub 现有模式。

```cpp
common::Px4Setpoint PublishFromIntent(const common::ControlIntent& intent) {
    double vx = intent.target_x, vy = intent.target_y, vz = intent.target_z;
    if (intent.type == common::ControlIntentType::kVelocityHeading) {
        // 速度幅值限幅（等比，方向不变）
        const double speed = std::sqrt(vx * vx + vy * vy + vz * vz);
        if (speed > config_.max_speed_mps) {
            const double scale = config_.max_speed_mps / speed;
            vx *= scale; vy *= scale; vz *= scale;
            ++clamped_count_;
        }
        // 无扰衔接：来源/类型变化时斜坡起点=上一周期实际输出
        if (intent.control_source != last_source_ || intent.type != last_type_) {
            vx_slew_ = SlewRateLimiter(config_.max_accel_mps2, last_output_.x);
            vy_slew_ = SlewRateLimiter(config_.max_accel_mps2, last_output_.y);
            vz_slew_ = SlewRateLimiter(config_.max_accel_mps2, last_output_.z);
        }
        // dt<=0（时钟回退/首拍）时 SlewRateLimiter 冻结输出——既有语义，勿改
        const double dt_s = config_.period.count() / 1000.0;
        common::Px4Setpoint out;
        out.type = common::SetpointType::kVelocity;
        out.x = static_cast<float>(vx_slew_.Limit(vx, dt_s));
        out.y = static_cast<float>(vy_slew_.Limit(vy, dt_s));
        out.z = static_cast<float>(vz_slew_.Limit(vz, dt_s));
        // 航向通道透传（互斥约束沿用 Px4Link 现有校验）
        out.use_yaw = intent.yaw_deg != 0.f;
        out.yaw_deg = intent.yaw_deg;
        out.use_yaw_rate = intent.yaw_rate_dps != 0.f;
        out.yaw_rate_dps = intent.yaw_rate_dps;
        out.valid = true;
        last_source_ = intent.control_source;
        last_type_ = intent.type;
        last_output_ = out;
        return out;
    }
    // 位置式（S5 远端引导）：直接限幅后透传位置目标，速度上限写入由
    // Px4Link 侧 type mask 语义保证；位置目标本身不做 slew（位置阶跃由
    // PX4 内部轨迹规划平滑，本控制器只管速度式连续性）
    ...
}
```

```cpp
bool AuthorityGateOpen(uint64_t now_ms) {
    if (px4_link_ == nullptr) return false;                 // 未注入（影子模式）
    const auto state = px4_link_->LatestState();
    const bool offboard = state.connected && state.armed &&
                          state.flight_mode == 6 && state.flight_sub_mode == 0;
    if (!offboard) return false;
    return state_machine_requested_;  // 状态机显式请求标志（Task 7 经意图 reason_code 传入）
}

void Tick(uint64_t now_ms) {
    DrainIntents();
    const bool gate = AuthorityGateOpen(now_ms);
    switch (authority_) {
        case ControlAuthority::kIdle:
            if (gate) { authority_ = ControlAuthority::kActive; SPDLOG_WARN(...); }
            else return;
            break;
        case ControlAuthority::kActive:
            if (!gate) { authority_ = ControlAuthority::kIdle; PublishInvalidStop(); SPDLOG_WARN(...); return; }
            break;
        case ControlAuthority::kCoast:
            if (!gate) { authority_ = ControlAuthority::kIdle; PublishInvalidStop(); return; }
            if (now_ms - coast_since_ms_ > config_.coast_timeout.count()) {
                authority_ = ControlAuthority::kIdle; PublishInvalidStop(); SPDLOG_WARN(...); return;
            }
            if (FreshIntentAvailable(now_ms)) authority_ = ControlAuthority::kActive;
            else { PublishBrake(); return; }
            break;
    }
    // ... 门禁③有限性、①超龄（超龄→kCoast 记录 coast_since_ms_）→ 限幅流水线 → 发布
}
```
